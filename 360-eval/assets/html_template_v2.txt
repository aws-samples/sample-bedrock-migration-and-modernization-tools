    // Reading Progress Bar
    window.addEventListener('scroll', function() {
      const winScroll = document.body.scrollTop || document.documentElement.scrollTop;
      const height = document.documentElement.scrollHeight - document.documentElement.clientHeight;
      const scrolled = (winScroll / height) * 100;
      document.getElementById('progressBar').style.width = scrolled + '%';
    });

    // Smooth scroll to section
    function scrollToSection(sectionId) {
      const element = document.getElementById(sectionId);
      if (element) {
        const navHeight = document.querySelector('.nav').offsetHeight;
        const elementPosition = element.getBoundingClientRect().top + window.pageYOffset;
        window.scrollTo({
          top: elementPosition - navHeight - 20,
          behavior: 'smooth'
        });
      }
    }

    // Task accordion toggle
    function toggleTaskAccordion(taskKey) {
      const accordion = document.getElementById('accordion-' + taskKey);
      if (accordion) {
        accordion.classList.toggle('active');
        // Trigger resize for any Plotly charts inside
        setTimeout(function() {
          window.dispatchEvent(new Event('resize'));
        }, 100);
      }
    }

    // Expand all task accordions
    function expandAllTasks() {
      document.querySelectorAll('.task-accordion').forEach(function(el) {
        el.classList.add('active');
      });
      window.dispatchEvent(new Event('resize'));
    }

    // Collapse all task accordions
    function collapseAllTasks() {
      document.querySelectorAll('.task-accordion').forEach(function(el) {
        el.classList.remove('active');
      });
    }

    // TTFB task switcher
    function showTTFBTask(taskName) {
      document.querySelectorAll('.ttfb-task-chart').forEach(function(el) {
        el.style.display = 'none';
      });
      const selectedChart = document.getElementById('ttfb-hist-' + taskName);
      if (selectedChart) {
        selectedChart.style.display = 'block';
        window.dispatchEvent(new Event('resize'));
      }
    }

    // Accuracy task switcher
    function showAccuracyTask(taskName) {
      document.querySelectorAll('.accuracy-task-chart').forEach(function(el) {
        el.style.display = 'none';
      });
      const selectedChart = document.getElementById('accuracy-hist-' + taskName);
      if (selectedChart) {
        selectedChart.style.display = 'block';
        window.dispatchEvent(new Event('resize'));
      }
    }

    // Output tokens task switcher
    function showOutputTokensTask(taskName) {
      document.querySelectorAll('.output-tokens-task-chart').forEach(function(el) {
        el.style.display = 'none';
      });
      const selectedChart = document.getElementById('output-tokens-hist-' + taskName);
      if (selectedChart) {
        selectedChart.style.display = 'block';
        window.dispatchEvent(new Event('resize'));
      }
    }

    window.addEventListener('load', function() {
      window.dispatchEvent(new Event('resize'));
    });
